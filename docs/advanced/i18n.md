# 国际化

国际化（Internationalization，简称i18n）是Mars Admin的重要特性之一，支持多语言切换和本地化配置。

## 功能概述

Mars Admin的国际化模块提供了以下功能：

- **多语言支持**：支持中文和英文两种语言
- **动态切换**：运行时动态切换语言
- **组件国际化**：Ant Design Vue组件的国际化
- **日期国际化**：Day.js日期库的国际化
- **持久化存储**：语言设置的本地存储
- **类型安全**：完整的TypeScript类型支持

## 技术架构

### 1. 国际化框架

Mars Admin使用Vue I18n作为国际化解决方案：

```typescript
// src/locales/index.ts
import { createI18n } from 'vue-i18n';
import { localStg } from '@/utils/storage';
import messages from './locale';

const i18n = createI18n({
  locale: localStg.get('lang') || 'zh-CN',
  fallbackLocale: 'en',
  messages,
  legacy: false
});

export function setupI18n(app: App) {
  app.use(i18n);
}

export const $t = i18n.global.t as App.I18n.$T;

export function setLocale(locale: App.I18n.LangType) {
  i18n.global.locale.value = locale;
}
```

## 组件国际化

### 1. Ant Design Vue国际化

```typescript
// src/locales/antd.ts
import type { Locale } from 'ant-design-vue/es/locale';
import zhCN from 'ant-design-vue/es/locale/zh_CN';
import enUS from 'ant-design-vue/es/locale/en_US';

export const antdLocales: Record<App.I18n.LangType, Locale> = {
  'zh-CN': zhCN,
  'en-US': enUS
};
```

在App.vue中使用：

```vue
<script setup lang="ts">
import { computed } from 'vue';
import { ConfigProvider } from 'ant-design-vue';
import { useAppStore } from './store/modules/app';
import { antdLocales } from './locales/antd';

const appStore = useAppStore();

const antdLocale = computed(() => {
  return antdLocales[appStore.locale];
});
</script>

<template>
  <ConfigProvider :locale="antdLocale">
    <router-view />
  </ConfigProvider>
</template>
```

### 2. Day.js国际化

```typescript
// src/locales/dayjs.ts
import { locale } from 'dayjs';
import 'dayjs/locale/zh-cn';
import 'dayjs/locale/en';
import { localStg } from '@/utils/storage';

export function setDayjsLocale(lang: App.I18n.LangType = 'zh-CN') {
  const localMap = {
    'zh-CN': 'zh-cn',
    'en-US': 'en'
  } satisfies Record<App.I18n.LangType, string>;

  const l = lang || localStg.get('lang') || 'zh-CN';

  locale(localMap[l]);
}
```

## 语言切换功能

### 1. 语言切换组件

```vue
<!-- src/components/common/lang-switch.vue -->
<script setup lang="ts">
import { computed } from 'vue';
import { $t } from '@/locales';

interface Props {
  lang: App.I18n.LangType;
  langOptions: App.I18n.LangOption[];
  showTooltip?: boolean;
}

const props = withDefaults(defineProps<Props>(), {
  showTooltip: true
});

type Emits = {
  (e: 'changeLang', lang: App.I18n.LangType): void;
};

const emit = defineEmits<Emits>();

const tooltipContent = computed(() => {
  if (!props.showTooltip) return '';
  return $t('icon.lang');
});

function changeLang(lang: App.I18n.LangType) {
  emit('changeLang', lang);
}
</script>

<template>
  <ADropdown placement="bottom">
    <ButtonIcon :tooltip-content="tooltipContent" tooltip-placement="left">
      <SvgIcon icon="heroicons:language" />
    </ButtonIcon>
    <template #overlay>
      <AMenu :selected-keys="[lang]">
        <AMenuItem v-for="option in langOptions" :key="option.key" @click="changeLang(option.key)">
          {{ option.label }}
        </AMenuItem>
      </AMenu>
    </template>
  </ADropdown>
</template>
```

### 2. 状态管理

```typescript
// src/store/modules/app/index.ts
import { ref } from 'vue';
import { setLocale } from '@/locales';
import { setDayjsLocale } from '@/locales/dayjs';
import { localStg } from '@/utils/storage';

const locale = ref<App.I18n.LangType>(localStg.get('lang') || 'zh-CN');

const localeOptions: App.I18n.LangOption[] = [
  {
    label: '中文',
    key: 'zh-CN'
  },
  {
    label: 'English',
    key: 'en-US'
  }
];

function changeLocale(lang: App.I18n.LangType) {
  locale.value = lang;
  setLocale(lang);
  setDayjsLocale(lang);
  localStg.set('lang', lang);
}

function updateDocumentTitleByLocale() {
  const { i18nKey, title } = router.currentRoute.value.meta;
  const documentTitle = i18nKey ? $t(i18nKey) : title;
  useTitle(documentTitle);
}
```

## 常见问题

### Q: 如何处理复数形式？

**解决方案：**
```typescript
// 使用Vue I18n的复数规则
const messages = {
  'zh-CN': {
    item: '{count} 个项目'
  },
  'en-US': {
    item: '{count} item | {count} items'
  }
};

// 使用
$t('item', { count: 1 }); // "1 item"
$t('item', { count: 5 }); // "5 items"
```

### Q: 如何处理日期和数字格式化？

**解决方案：**
```typescript
// 使用Intl API
const formatNumber = (num: number, locale: string) => {
  return new Intl.NumberFormat(locale).format(num);
};

const formatDate = (date: Date, locale: string) => {
  return new Intl.DateTimeFormat(locale).format(date);
};
```

### Q: 如何实现RTL语言支持？

**解决方案：**
```typescript
// 检测RTL语言
const isRTL = (locale: string) => {
  const rtlLanguages = ['ar', 'he', 'fa'];
  return rtlLanguages.some(lang => locale.startsWith(lang));
};

// 应用RTL样式
if (isRTL(currentLocale)) {
  document.documentElement.dir = 'rtl';
  document.documentElement.classList.add('rtl');
}
```
